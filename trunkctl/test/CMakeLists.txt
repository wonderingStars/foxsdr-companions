# Pinned expectations are in expected/*.jsonl; to rewrite them after an intended change run
#   TC_UPDATE=1 ctest ... and review the git diff.

# tc(<name> <mode> [RATE hz] [FMT fmt] [SIG scenario] [SIGOPT opt...] [ARGS helper args...] [EXPECT file]
#    [EXTRA -Dvar=value...])
# The helper is started with --foxsdr-helper 1 --format FMT --rate RATE then ARGS (except in mode fail, where ARGS
# is the whole command line). The signal is made by trunkgen: SIG at RATE in FMT with the SIGOPT options.
function(tc NAME MODE)
  cmake_parse_arguments(C "" "RATE;FMT;SIG;EXPECT" "SIGOPT;ARGS;EXTRA" ${ARGN})
  if(NOT C_RATE)
    set(C_RATE 48000)
  endif()
  if(NOT C_FMT)
    set(C_FMT s16)
  endif()
  if(NOT C_EXPECT)
    set(C_EXPECT ${NAME})
  endif()
  if(MODE STREQUAL "fail")
    set(all ${C_ARGS})
  else()
    set(all --foxsdr-helper 1 --format ${C_FMT} --rate ${C_RATE} ${C_ARGS})
  endif()
  string(REPLACE ";" "|" a "${all}")
  set(defs -DMODE=${MODE} -DNAME=${NAME} -DHELPER=$<TARGET_FILE:trunkctl> "-DARGS=${a}"
      -DSIGTOOL=$<TARGET_FILE:trunkctl_sigtool> -DTRUNKGEN=$<TARGET_FILE:trunkgen> -DWORK=${WORKDIR}
      -DEXPECTED=${TE}/${C_EXPECT}.jsonl)
  if(MODE STREQUAL "info")
    list(APPEND defs -DEXPECTED=${PROJECT_SOURCE_DIR}/trunkctl.helper.json) # later -D wins
  endif()
  if(C_SIG)
    string(REPLACE ";" "|" g "${C_SIG};${C_RATE};${C_FMT};${C_SIGOPT}")
    list(APPEND defs "-DGEN=${g}")
  endif()
  add_test(NAME ${NAME} COMMAND ${CMAKE_COMMAND} ${defs} ${C_EXTRA} -P ${RUN})
endfunction()
